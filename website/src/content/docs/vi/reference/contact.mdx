import { Tabs, TabItem } from "@astrojs/starlight/components";

`client.contacts` quản lý các liên hệ — khách hàng tiềm năng, khách hàng, hoặc người dùng giao tiếp qua các kênh của bạn. Mỗi liên hệ có thể có các hội thoại, comments, files, và lịch sử hoạt động liên kết.

---

## Schema

### Contact

| Trường | Kiểu | Mô tả |
|-------|------|-------|
| `object_name` | string? | Định danh loại đối tượng |
| `id` | string | ID duy nhất của liên hệ |
| `organization_id` | string | Tổ chức sở hữu liên hệ này |
| `display_name` | string? | Tên hiển thị của liên hệ |
| `email` | string? | Địa chỉ email |
| `phone_number` | string? | Số điện thoại |
| `avatar_url` | string? | URL ảnh đại diện |
| `created_at` | string | ISO 8601 thời gian tạo |
| `updated_at` | string | ISO 8601 thời gian cập nhật |

### UpdateContactInput

| Trường | Kiểu | Bắt buộc | Mô tả |
|-------|------|----------|-------|
| `name` | string | | Cập nhật tên hiển thị |
| `email` | string | | Cập nhật địa chỉ email |
| `phone` | string | | Cập nhật số điện thoại |

### ContactComment

| Trường | Kiểu | Mô tả |
|-------|------|-------|
| `_id` | string | ID duy nhất của comment |
| `text` | string? | Nội dung comment |
| `created_at` | string? | ISO 8601 thời gian tạo |

### ContactFile

| Trường | Kiểu | Mô tả |
|-------|------|-------|
| `_id` | string | ID duy nhất của file |
| `name` | string? | Tên file |
| `url` | string? | URL tải xuống |
| `size` | number? | Kích thước file (bytes) |

## Phương thức

| Phương thức | TypeScript | Python | Mô tả |
|--------|-----------|--------|-------|
| Danh sách | `list` | `list` | Danh sách liên hệ có phân trang |
| Lấy | `get` | `get` | Lấy liên hệ theo ID |
| Cập nhật | `update` | `update` | Cập nhật trường liên hệ |
| Tìm kiếm | `search` | `search` | Tìm kiếm toàn văn trên các liên hệ |
| Xuất CSV | `exportCsv` | `export_csv` | Tải xuống tất cả liên hệ dưới dạng CSV |
| Lấy hội thoại | `getConversations` | `get_conversations` | Danh sách hội thoại của liên hệ |
| Lấy comment | `getComments` | `get_comments` | Danh sách comments nội bộ trên liên hệ |
| Lấy file | `getFiles` | `get_files` | Danh sách files đính kèm với liên hệ |
| Lấy hoạt động | `getActivities` | `get_activities` | Nhật ký hoạt động hội thoại |
| Tải lên avatar | `uploadAvatar` | `upload_contacts` | Tải lên file cho liên hệ (avatar / import file) |
| Danh sách thông báo | `listNotifications` | `list_notifications` | Danh sách thông báo cho người dùng hiện tại |
| Đánh dấu thông báo đã đọc | `markNotificationsRead` | `mark_notifications_read` | Đánh dấu thông báo đã đọc |
| Bỏ qua thông báo | `dismissNotification` | `dismiss_notification` | Bỏ qua một thông báo |
| Bỏ qua tất cả thông báo | `dismissAllNotifications` | `dismiss_all_notifications` | Xóa tất cả thông báo |

## list / list

<Tabs syncKey="lang">
<TabItem label="TypeScript">
```typescript
const page = await client.contacts.list({ limit: 50, skip: 0 });
for (const contact of page.data) {
  console.log(contact.id, contact.display_name, contact.email);
}
```
</TabItem>
<TabItem label="Python">
```python
page = client.contacts.list(limit=50, skip=0)
for contact in page.get("data", []):
    print(contact["id"], contact.get("display_name"), contact.get("email"))
```
</TabItem>
